cmake_minimum_required(VERSION 3.24)
project(AetherMidiEvaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(AETHER_ORT_ROOT "" CACHE PATH "Verified existing ONNX Runtime 1.31.0 SDK")
if(NOT EXISTS "${AETHER_ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Supply an existing verified SDK with AETHER_ORT_ROOT; this tool downloads no dependencies")
endif()
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
    set(ort_library "${AETHER_ORT_ROOT}/lib/libonnxruntime.1.31.0.dylib")
elseif(WIN32)
    set(ort_library "${AETHER_ORT_ROOT}/lib/onnxruntime.lib")
else()
    message(FATAL_ERROR "Evaluation currently supports macOS arm64 and Windows x64")
endif()
add_executable(AetherMidiEvaluate main.cpp)
target_include_directories(AetherMidiEvaluate SYSTEM PRIVATE "${AETHER_ORT_ROOT}/include")
target_link_libraries(AetherMidiEvaluate PRIVATE "${ort_library}")
if(APPLE)
    add_executable(AetherMidiAudition audition_mac.cpp)
    target_link_libraries(AetherMidiAudition PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
    target_compile_options(AetherMidiAudition PRIVATE -Wall -Wextra -Wpedantic)
elseif(WIN32)
    target_link_libraries(AetherMidiEvaluate PRIVATE psapi bcrypt)
endif()
if(MSVC)
    target_compile_options(AetherMidiEvaluate PRIVATE /W4 /permissive-)
    add_custom_command(TARGET AetherMidiEvaluate POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${AETHER_ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:AetherMidiEvaluate>/onnxruntime.dll")
else()
    target_compile_options(AetherMidiEvaluate PRIVATE -Wall -Wextra -Wpedantic)
endif()
